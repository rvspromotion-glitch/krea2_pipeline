# Hard requirements of the node set, not preferences.
#
# numpy 2 and mediapipe above 0.10.14 both break LayerStyle and Impact-Pack at
# import time. Letting pip resolve these freely yields an image that builds
# clean and then fails on the first render, which is the expensive place to
# find out. Applied as a pip constraint file across every install in the build,
# so a node package's own requirements.txt cannot walk them back.
# Matches the detailer worker's working set, which runs these nodes in
# production — transformers is floored higher there than the 4.39 this used to
# ask for, and headless opencv avoids pulling GUI libs into a worker.
numpy<2
protobuf<5
transformers>=4.45.0
safetensors
mediapipe==0.10.14

# contrib, not plain: LayerStyle imports guidedFilter from cv2.ximgproc, which
# only the contrib build ships. Without it every boot logs "Cannot import name
# 'guidedFilter' from 'cv2.ximgproc'" and a handful of its nodes are quietly
# broken. It is a strict superset of opencv-python-headless, and headless still,
# so no GUI libs come along.
opencv-contrib-python-headless

# Not load-bearing. Both graphs carry a PathchSageAttentionKJ node, but the
# worker strips it (see BYPASS_CLASSES in src/graph.py) because its Triton
# kernels abort on Blackwell. Kept installed only so KREA2_SAGE_ATTENTION=1
# works on an endpoint whose cards do support it.
sageattention
